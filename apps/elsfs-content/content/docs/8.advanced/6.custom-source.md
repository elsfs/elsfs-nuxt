---
title: 自定义数据源
description: 定义自定义数据源以获取数据。
---

默认情况下，Nuxt Content 提供了一些内置数据源，例如本地文件数据源和远程 GitHub 数据源。但在某些场景下这还不够，比如你想从远程 API 获取数据。这时，你可以定义一个自定义数据源来获取数据，并在集合中使用它。

使用 `defineCollectionSource`，你可以定义自定义数据源。

```ts
import { defineCollectionSource } from '@nuxt/content'

const hackernewsSource = defineCollectionSource({
  getKeys: () => {
    return fetch('https://hacker-news.firebaseio.com/v0/topstories.json')
      .then(res => res.json())
      .then(data => data.map((key: string) => `${key}.json`))
  },
  getItem: (key: string) => {
    const id = key.split('.')[0]
    return fetch(`https://hacker-news.firebaseio.com/v0/item/${id}.json`)
      .then(res => res.json())
  },
})
```

然后你就可以在集合中使用这个数据源。

```ts [content.config.ts]
import { defineContentConfig, defineCollectionSource, defineCollection } from '@nuxt/content'
import { z } from 'zod'

const hackernewsSource = defineCollectionSource({
  getKeys: () => {
    return fetch('https://hacker-news.firebaseio.com/v0/topstories.json')
      .then(res => res.json())
      .then(data => data.map((key: string) => `${key}.json`))
  },
  getItem: (key: string) => {
    const id = key.split('.')[0]
    return fetch(`https://hacker-news.firebaseio.com/v0/item/${id}.json`)
      .then(res => res.json())
  },
})

const hackernews = defineCollection({
  type: 'data',
  source: hackernewsSource,
  schema: z.object({
    title: z.string(),
    date: z.date(),
    type: z.string(),
    score: z.number(),
    url: z.string(),
    by: z.string(),
  }),
})

export default defineContentConfig({
  collections: {
    hackernews,
  },
})
```
